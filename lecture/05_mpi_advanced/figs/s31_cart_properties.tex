% What a Cartesian communicator is made of: a number of dimensions, a size and
% a periodicity flag per dimension, and a rank order that maps coordinates to
% rank numbers.

\input{_cart.tex}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

  \carttopo{0/0/h, 1/0/h, 2/0/h, 3/0/h, 0/1/h, 1/1/h, 2/1/h, 3/1/h,
            0/2/h, 1/2/h, 2/2/h, 3/2/h, 0/3/h, 1/3/h, 2/3/h, 3/3/h}

  % rank numbering runs row by row
  \foreach \c/\l in {0/0, 1/1, 2/2}{
    \node[text=inkdark,font=\sffamily\tiny] at ({\c*\cartd},0) {\l};}

  \newcommand{\ann}[5]{%
    \draw[->,draw=blu,line width=0.8pt] (#1,#2) -- (#3,#4);
    \node[text=fg,font=\sffamily\scriptsize,align=left,anchor=west] at (#1,#2) {#5};}

  \node[text=fg,font=\sffamily\scriptsize,anchor=south east] at (0.20,1.02) {rank order};
  \draw[->,draw=blu,line width=0.8pt] (0.24,0.96) -- (0.06,0.22);

  %\ann{3.95}{0.95}{2.55}{0.52}{number of dimensions}
  \ann{2.95}{-1.55}{2.8}{-1.0}{periodicity (per dimension)}
  \ann{3.95}{-2.45}{1.6}{-1.6}{rank at coordinates 2,2}

  \draw[decorate,decoration={brace,amplitude=4pt},draw=blu,line width=0.7pt]
    (-0.62,{-3*\cartd}) -- (-0.62,0);
  \node[text=fg,font=\sffamily\scriptsize,anchor=east]
    at (-0.82,{-1.5*\cartd}) {size (per dimension)};

\end{tikzpicture}
