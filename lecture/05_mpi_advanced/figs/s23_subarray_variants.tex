% Receiving a subarray into a full-size array (left) writes only the sub-block;
% receiving it into an array of exactly SUBSIZE (right) fills that completely.
\documentclass[dvisvgm,border=3pt]{standalone}
\input{preamble.tex}
\input{mem.tex}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

  \def\cs{0.40}

  \memgrid{(0,0)}{8}{8}{\cs}
  \memblock{(0,0)}{2}{2}{3}{3}{\cs}
  \draw[decorate,decoration={brace,amplitude=4pt},draw=blu,line width=0.7pt]
    (-0.14,{-8*\cs}) -- (-0.14,0);
  \node[text=fg,font=\sffamily\scriptsize,anchor=east] at (-0.34,{-4*\cs}) {SIZE};

  \draw[->,draw=blu,line width=0.9pt]
    ({8*\cs+1.55},{-6.3*\cs}) -- ({5.4*\cs},{-5.6*\cs});
  \node[text=fg,font=\sffamily\scriptsize,align=left,anchor=west]
    at ({8*\cs+1.60},{-6.3*\cs}) {unwritten part\\of new array};

  \begin{scope}[xshift=7.4cm,yshift=-1.0cm]
    \memgrid{(0,0)}{3}{3}{\cs}
    \memblock{(0,0)}{0}{0}{3}{3}{\cs}
    \draw[decorate,decoration={brace,amplitude=4pt},draw=blu,line width=0.7pt]
      ({3*\cs+0.14},0) -- ({3*\cs+0.14},{-3*\cs});
    \node[text=fg,font=\sffamily\scriptsize,anchor=west]
      at ({3*\cs+0.34},{-1.5*\cs}) {SUBSIZE};
  \end{scope}

\end{tikzpicture}
\end{document}
